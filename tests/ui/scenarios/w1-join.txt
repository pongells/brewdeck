# W1 join a network with a password typed on the screen: the core gets the join with that name and password.
# checks: W1
wait 500
wait_for screen.ready 10000
tap_tag action.menu
wait 800
tap_tag action.wifi_settings
expect_visible screen.wifi 2000
expect_visible row.wifi.Home Wi-Fi 2 4000
tap_tag row.wifi.Home Wi-Fi 2
expect_visible screen.password 2000
type input.password Coffee-42
wait 500
shot password typed
wait 300
tap_tag action.join
wait 500
expect_cmd wifi_join ssid="Home Wi-Fi 2" pass_len=9
quit
