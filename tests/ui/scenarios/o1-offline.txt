# O4 a short drop goes unnoticed; O1 after 5 s offline the screen says what it is doing, after 20 s what is wrong;
# O3 a way out for each case; O5 back to Ready as soon as the machine answers.
# checks: O1, O3, O4, O5
wait 500
wait_for screen.ready 10000
# a 3 s drop
button Machine on/off
wait 3000
expect_hidden screen.offline 100
expect_text value.state Ready|Heating|Cooling|Warming 100
button Machine on/off
wait 3000
# the machine gone
button Machine on/off
expect_visible screen.offline 7000
wait 2000
shot offline: looking
button +20 s
wait 1500
expect_visible action.machine_address
expect_visible action.wifi_settings
shot offline: machine not found
button Machine on/off
expect_hidden screen.offline 3000
expect_visible screen.ready
wait 2000
# the Wi-Fi gone
button Wi-Fi on/off
expect_visible screen.offline 7000
button +20 s
wait 1500
expect_visible action.retry
expect_visible action.wifi_settings
shot offline: no Wi-Fi
button Wi-Fi on/off
expect_hidden screen.offline 8000
quit
